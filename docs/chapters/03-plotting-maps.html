4  Plotting & Maps

4.1 Basic Plotting with Plots.jl

using Plots

x = 0:0.1:2π
y = sin.(x)
plot(x, y, label="sin(x)", xlabel="x", ylabel="y", lw=2)

4.1.1 Multiple Series

plot(x, sin.(x), label="sin(x)", lw=2)
plot!(x, cos.(x), label="cos(x)", lw=2, ls=:dash)

4.1.2 Scatter Plots

x = randn(50)
y = randn(50)
scatter(x, y, xlabel="X", ylabel="Y", title="Random Points", 
        markersize=5, alpha=0.7)

4.1.3 Heatmaps

z = [sin(x) * cos(y) for x in 0:0.1:2π, y in 0:0.1:2π]
heatmap(z, xlabel="X", ylabel="Y", title="2D Function", 
        color=:viridis)

4.2 Subplots

p1 = plot(x, sin.(x), title="Sine")
p2 = plot(x, cos.(x), title="Cosine")
p3 = plot(x, tan.(x), title="Tangent", ylims=(-5, 5))
p4 = scatter(randn(20), randn(20), title="Random")

plot(p1, p2, p3, p4, layout=(2, 2), size=(700, 500))

4.3 Contour Plots

x = -2:0.1:2
y = -2:0.1:2
z = [exp(-(xi^2 + yi^2)) for xi in x, yi in y]

contour(x, y, z', levels=10, fill=true, 
        xlabel="X", ylabel="Y", title="Gaussian")

4.4.1 Coordinate Systems

# Example with projected coordinates
using CoordinateTransformations
using Proj4

# Transform from WGS84 to UTM
wgs84 = Proj4.Projection("+proj=longlat +datum=WGS84")
utm35 = Proj4.Projection("+proj=utm +zone=35 +datum=WGS84")

4.4.2 Plotting Geophysical Data

# Gravity anomaly map example
using Plots

# Create synthetic gravity data
nx, ny = 50, 50
x = range(0, 10, length=nx)
y = range(0, 10, length=ny)
gz = [10 * exp(-((xi-5)^2 + (yi-5)^2)/2) for xi in x, yi in y]

heatmap(x, y, gz', 
        xlabel="Easting (km)", 
        ylabel="Northing (km)",
        title="Bouguer Gravity Anomaly",
        colorbar_title="mGal",
        color=:RdBu)
